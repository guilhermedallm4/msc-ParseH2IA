\section{Classes de Dependencia}
\label{cap:classe_deprel}
\begin{itemize}

\item \textbf{acl}: Uma oração finita ou não finita que modifica um substantivo. \textbf{Exemplo}: as questões como ele as vê. acl(questões, vê). Variação de \textbf{acl} é \textbf{acl:relcl}, modificador de oração relativa.

\item\textbf{advcl}: Uma cláusula que modifica um verbo ou outro predicado como modificador, não como complemento central. \textbf{Exemplo}: O acidente aconteceu enquanto a noite caía. advcl(aconteceu, caía). Variação de \textbf{advcl} é \textbf{advcl:relcl}, modificador de oração adverbial relativa.

\item\textbf{advmod}: Um advérbio ou frase adverbial que modifica um predicado ou modificador. \textbf{Exemplo}: Alimento geneticamente modificado. advmod(modificado, geneticamente). Variações de \textbf{advmod} são, \textbf{advmod:emph}, palavra enfatizadora, itensificador e \textbf{advmod:lmod} modificador de adverbial locativo.

\item\textbf{amod}: Uma frase adjetiva que modifica um substantivo. \textbf{Exemplo}: Pedro come cachorros-quentes grandes. amod(cachorros-quentes, grandes).

\item\textbf{appos}: Um nominal que define ou descreve outro nominal imediatamente após. \textbf{Exemplo}: Tiago, meu irmão, chegou. appos(Tiago, irmão).

\item\textbf{aux}: Uma palavra funcional associada a um verbo léxico que expressa categorias gramaticais como tempo ou modo. \textbf{Exemplo}: Pedro já chegou. aux(chegou, já). Variação de \textbf{aux} é \textbf{aaux:pass}, auxiliar passivo, Tiago foi embora. aux:pass(embora, foi).

\item\textbf{case}: Uma palavra funcional que indica o caso gramatical de um nominal. \textbf{Exemplo}: O escritório do presidente. case(presidente, do).

\item\textbf{cc}: Uma palavra que liga palavras ou constituintes maiores. \textbf{Exemplo}: Pedro é grande e honesto. cc(é, e). Variação de \textbf{cc} é \textbf{cc:preconj}, um elemento antes de uma conjunção coordenativa. \textbf{Exemplo}: Tanto os meninos quanto as meninas estão aqui. cc:preconj(meninos, tanto) 

\item\textbf{ccomp}: Uma oração dependente que completa o significado de um verbo. \textbf{Exemplo}: Ele diz que você gosta de nadar. ccomp(diz, gosta)

\item\textbf{clf}: Uma palavra que categoriza um substantivo, usada com numerais. \textbf{Exemplo}: Três unidades de lapís. clf(lapís, unidades).

\item\textbf{compound}: Um termo que forma um composto com sua cabeça. \textbf{Exemplo}: Livro telefônico. compound(livro, telefônico). Variações de \textbf{compound} são, \textbf{compound:lvc}, um verbo auxiliar em uma construção de verbo leve. \textbf{Exemplo}: Eles preferem o trem. compound: lvc(preferem, o); \textbf{compound:prt}, partícula de verbo frasal. \textbf{Exemplo}: Eles desligaram a estação. compound:prt(desligaram, desligar); \textbf{compound:redup}, elementos em compostos por reduplicação. \textbf{Exemplo}: Homens grandes (+ênfase) estão jogando jogos. compound:redup(grandes, ênfase); \textbf{compound:svc}, compostos de verbos seriais. \textbf{Exemplo}: Musa veio pegar a faca. compound:svc(veio, pegar).

\item\textbf{conj}: Um coordenado ligado por conjunção. \textbf{Exemplo}: Pedro é grande e honesto. conj(é, honesto).

\item\textbf{cop}: Uma palavra que liga um sujeito a um predicado, tipicamente ``ser''. \textbf{Exemplo}: Tiago é honesto. cop(é, é).

\item\textbf{csubj}: Um oracional que funciona como sujeito sintático. \textbf{Exemplo}: Que ele mentiu me surpreendeu. csubj(surpreendeu, mentiu). As variações de \textbf{csubj} são, \textbf{csubj:outer}, sujeito de uma oração copulativa cujo predicado é uma oração. \textbf{Exemplo}: Caminhar nas montanhas é experimentar o melhor da natureza. csubj:outer(é, caminhar) e \textbf{csubj:pass} sujeito passivo de uma oração. \textbf{Exemplo}: Que ela mentiu foi suspeitado por todos. csubj:pass(for, mentiu).

\item\textbf{dep}: Dependência não especificada. Usado quando nenhuma outra etiqueta se aplica. \textbf{Exemplo}: Meu amigo não faz isso tão bem. dep(faz, não).

\item\textbf{det}: Determinante. Uma palavra que introduz uma frase nominal e expressa noção gramatical como definitude. \textbf{Exemplo}: O homem está aqui. det(está, o). As variações de \textbf{det} são, \textbf{det:numgov}] quantificador pronominal que governa o caso do substantivo. \textbf{Exemplo}: Quantos homens jogaram cartas. det:numgov(homens, quantos); \textbf{det:nummod} quantificador pronominal que concorda em caso com o substantivo. \textbf{Exemplo}: Com quantos homens eu joguei cartas. det:nummod(homens, quantos); \textbf{det:poss} determinante possessivo. \textbf{Exemplo}: Vai ser meu cuidado verificar. det:poss(é, meu).

\item\textbf{discourse}: Um marcador discursivo ou conector. \textbf{Exemplo}: Eu tenho 21 anos, bem, vou fazer em novembro. discourse(tenho, bem).

\item\textbf{dislocated}: Elementos deslocados. Elementos no início ou fim de uma cláusula relacionados a uma posição interna. \textbf{Exemplo}: Elefante, tópico, nariz, sujeito, longo. dislocated(é, elefante).

\item\textbf{expl}: Expletivo. Um pronome sem conteúdo semântico em posição de argumento central. \textbf{Exemplo}: Há um fantasma no quarto. expl(há, há). As variações de \textbf{expl} são \textbf{expl:impers} Expletivo em construções impessoais. \textbf{Exemplo}: Espera-se que viaje. expl:impers(espera-se, se); \textbf{expl:pass} pronome reflexivo usado no passivo reflexivo. \textbf{Exemplo}: É dito facilmente. expl:pass(dito, é); \textbf{expl:pv} Clítico reflexivo com verbo inherentemente reflexivo. \textbf{Exemplo}: Martin teme animais. expl:pv(teme, se).

\item\textbf{fixed}: Uma expressão fixa composta de múltiplas palavras. \textbf{Exemplo}: Eu gosto de cachorros assim como de gatos. fixed(gosto, assim).

\item\textbf{flat}: Expressão plana. Uma expressão multi-palavra onde a cabeça não é clara. \textbf{Exemplo}: Hillary Rodham Clinton. flat(Hillary, Rodham). Algumas variações de \textbf{flat} são \textbf{flat:foreign} Termos estrangeiros em uma expressão plana. \textbf{Exemplo}: Azul no Rosto. flat:foreign(Azul, no) e \textbf{flat:name}, componentes de nomes próprios. \textbf{Exemplo}: Eis o discurso de Tiziana Maiolo. flat:name(Tiziana, Maiolo).

\item\textbf{goeswith}: Um espaço ou hífen inserido incorretamente. \textbf{Exemplo}: Eles vêm aqui sem permissão legal. goeswith(vêm, aqui).

\item\textbf{iobj}: Objeto indireto. O objeto indireto de um verbo. \textbf{Exemplo}: Ele me deu um aumento. iobj(deu, me).

\item\textbf{list}: Lista. Usado para cadeias de itens comparáveis \textbf{Exemplo}: Steve Jones sj@abc.xyz Universidade do Arizona. list(Steve, sj@abc.xyz).

\item\textbf{mark}: Uma palavra que marca uma cláusula como subordinada. \textbf{Exemplo}: Forças se envolveram em luta após insurgentes atacarem. mark(atacarem, após).

\item\textbf{nmod}: Modificador nominal. Dependentes nominais de outro nominal, como atributo ou complemento genitivo. \textbf{Exemplo}: O escritório da presidente. nmod(escritório, presidente). As variações de \textbf{nmod} são, \textbf{nmod:poss} modificador possessivo marcado com genitivo. \textbf{Exemplo}: O livro de Marie. nmod:poss(livro, Marie) e \textbf{nmod:tmod}: Modificador que especifica tempo. \textbf{Exemplo}: Você está livre para o almoço algum dia esta semana? nmod:tmod(dia, semana).

\item\textbf{nsubj}: Sujeito nominal. O sujeito sintático e proto-agente de uma cláusula. \textbf{Exemplo}: Clinton derrotou Dole. nsubj(derrotou, Clinton). As variações de \textbf{nsubj} são \textbf{nsubj:outer} sujeito de cláusula copulativa cujo predicado é cláusula. \textbf{Exemplo}: O problema é que isso nunca foi tentado. nsubj:outer(é, problema). \textbf{nsubj:pass}, frase nominal sujeito de cláusula passiva. \textbf{Exemplo}: Schwarzenberg foi derrotado por Zeman. nsubj:pass(foi, Schwarzenberg)

\item\textbf{nummod}: Um número que modifica um nominal. \textbf{Exemplo}: Sam comeu 3 ovelhas. nummod(comeu, 3). A variação do \textbf{nummod} é \textbf{nummod:gov}, modificador numérico que governa o caso do substantivo. \textbf{Exemplo}: Cinco homens jogaram cartas. nummod:gov(homens, cinco).

\item\textbf{obj}: O argumento central secundário de um verbo. \textbf{Exemplo}:: Ela me deu um aumento. obj(deu, aumento)

\item\textbf{obl}: Nominal oblíquo. Nominal não central atuando como argumento ou adjunto. \textbf{Exemplo}: Ontem à noite, nadei na piscina. obl(nadei, noite). As variações do \textbf{obl} são, \textbf{obl:agent} Agentes de verbos passivos.  \textbf{Exemplo}: O gato foi perseguido pelo cachorro. obl:agent(perseguido, cachorro). \textbf{obl:arg} Argumentos oblíquos distintos de adjuntos. \textbf{Exemplo}: Confio no instinto dele. obl:arg(confio, instinto). \textbf{obl:lmod} modificador especificando localização. \textbf{Exemplo}: Dirija por aquela estrada, você recebe instruções nela. obl:lmod(estrada, por). \textbf{obl:tmod}, modificador especificando tempo. \textbf{Exemplo}: Ontem à noite, nadei na piscina. obl:tmod(nadei, noite).

\item\textbf{orphan}: Usado em elipse de cabeça onde promoção resultaria em relação não natural. \textbf{Exemplo}: Marie ganhou ouro e Peter bronze. orphan(ganhou, Peter).

\item\textbf{parataxis}: Relação entre elementos justapostos sem coordenação explícita. \textbf{Exemplo}: Vamos encarar, estamos irritados. parataxis(estamos, encarar).

\item\textbf{punct} Qualquer peça de pontuação em uma cláusula. \textbf{Exemplo}: Vá para casa! punct(vá, !).

\item\textbf{reparandum} Indica disfluências em reparos de fala. \textbf{Exemplo}: Vá para a direi- para a esquerda. reparandum(esquerda, direi).

\item\textbf{root}: Aponta para a raiz da sentença. \textbf{Exemplo}: Eu amo batatas fritas francesas. root(amo, ROOT).

\item\textbf{vocative}: Marca um participante do diálogo endereçado. \textbf{Exemplo}: Rapazes, relaxem!. vocative(relaxem, rapazes).

\item\textbf{xcomp}: Complemento clausal aberto. Cláusula sem sujeito próprio, compartilhando com a cláusula matriz. \textbf{Exemplo}: Esperamos que eles mudem de ideia. xcomp(esperamos, mudem).

\end{itemize}
